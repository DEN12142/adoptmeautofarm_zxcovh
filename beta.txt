    local Players = game:GetService("Players")
    local UserInputService = game:GetService("UserInputService")

    local Player = Players.LocalPlayer

    if not Player.PlayerGui then
        repeat task.wait() until Player.PlayerGui
    end

    if _G.GCF_CONNS then
        for _, c in ipairs(_G.GCF_CONNS) do
            pcall(function() c:Disconnect() end)
        end

    _G.GCF_CONNS = {}
    _G.GCF_TOKEN = (_G.GCF_TOKEN or 0) + 1
    local myToken = _G.GCF_TOKEN

    local function track(conn)
        table.insert(_G.GCF_CONNS, conn)
        return conn
    end

    if old then old:Destroy() end

    -- Константы
    local GHOST_ATTR = {
        Id = "GhostClustersId",
        Size = "GhostClustersSize",
        Progress = "GhostClustersProgress",
        ProgressRequired = "GhostClustersProgressRequired",
        Shielded = "GhostClustersShielded",
        Orbiting = "GhostClustersOrbiting",
    }

    local autoTarget = true
    local flySpeed = 50
    local currentTargetIndex = 1
    local activeLoopId = 0

    -- ============================================
    -- ПОИСК ПРИЗРАКОВ
    -- ============================================

    local function get_alive_ghosts()
        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return {} end

        local visuals = workspace:FindFirstChild("GhostClustersVisuals")
        if not visuals then return {} end

        local ghosts = {}
        local playerPos = hrp.Position

        for _, model in ipairs(visuals:GetChildren()) do
            if model:IsA("Model") then
                local ghostId = model:GetAttribute(GHOST_ATTR.Id)
                if ghostId then
                    local isShielded = model:GetAttribute(GHOST_ATTR.Shielded) == true
                    local isOrbiting = model:GetAttribute(GHOST_ATTR.Orbiting) == true
                    local progress = model:GetAttribute(GHOST_ATTR.Progress) or 0
                    local required = model:GetAttribute(GHOST_ATTR.ProgressRequired) or 1

                    if not isShielded and not isOrbiting and progress < required then
                        local pos = model:GetPivot().Position
                        local dx = playerPos.X - pos.X
                        local dz = playerPos.Z - pos.Z
                        local dist = math.sqrt(dx * dx + dz * dz)

                        table.insert(ghosts, {
                            model = model,
                            id = ghostId,
                            size = model:GetAttribute(GHOST_ATTR.Size),
                            position = pos,
                            distance = dist,
                        })
                    end
                end
            end
        end

        table.sort(ghosts, function(a, b)
            return a.distance < b.distance
        end)

        return ghosts
    end

    local function fly_to(targetPos, dt)
        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end

        local currentPos = hrp.Position
        local direction = targetPos - currentPos
        local distance = direction.Magnitude

        if distance >= 3 then
            local moveVector = direction.Unit * math.min(flySpeed * dt, distance)
            hrp.CFrame = hrp.CFrame + moveVector
            hrp.Velocity = Vector3.new(0, 0, 0)
            hrp.RotVelocity = Vector3.new(0, 0, 0)
        end

        local p = hrp.Position
        local dx = targetPos.X - p.X
        local dz = targetPos.Z - p.Z
        return math.sqrt(dx * dx + dz * dz)
    end

    local function is_ghost_alive(ghost)
        if not ghost or not ghost.model or not ghost.model.Parent then
            return false
        end
        local progress = ghost.model:GetAttribute(GHOST_ATTR.Progress) or 0
        local required = ghost.model:GetAttribute(GHOST_ATTR.ProgressRequired) or 1
        return progress < required
    end

    local GREEN = Color3.fromRGB(80, 255, 80)
    local RED = Color3.fromRGB(255, 60, 60)
    local GRAY = Color3.fromRGB(150, 150, 150)
    local GREEN_BTN = Color3.fromRGB(60, 60, 80)
    local RED_BTN = Color3.fromRGB(160, 40, 40)

    local function update_status(text, color)
        StatusLabel.Text = text
        StatusLabel.TextColor3 = color or GRAY
        StatusDot.BackgroundColor3 = color or RED
    end

    local function update_target_info(size, dist)
        if size == nil then
            TargetInfoLabel.Text = "Target: None"
        else
            TargetInfoLabel.Text = string.format(
                "Target: %s | %d studs",
                tostring(size),
                math.floor(dist or 0)
            )
        end
    end

    local function set_farm_ui(isRunning, statusText, statusColor)
        if isRunning then
            ToggleButton.Text = "STOP FARM"
            ToggleButton.BackgroundColor3 = RED_BTN
        else
            ToggleButton.Text = "START FARM"
            ToggleButton.BackgroundColor3 = GREEN_BTN
        end
        update_status(statusText, statusColor)
    end

    -- ============================================
    -- ФАРМ ЦИКЛ
    -- ============================================

    local function farm_loop(myLoopId)
        while running and myLoopId == activeLoopId and _G.GCF_TOKEN == myToken do
            local waited = task.wait(0.03)
            local dt = (type(waited) == "number") and math.clamp(waited, 0.01, 0.1) or 0.03

            if not (running and myLoopId == activeLoopId and _G.GCF_TOKEN == myToken) then
                break
            end

            local okIter, errIter = pcall(function()
                local ghosts = get_alive_ghosts()
                GhostsLabel.Text = "Ghosts: " .. #ghosts

                local target
                if autoTarget then
                    target = ghosts[1]
                elseif #ghosts > 0 then
                    if currentTargetIndex > #ghosts then
                        currentTargetIndex = 1
                    end
                    target = ghosts[currentTargetIndex]
                end

                if target then
                    local distNow = fly_to(target.position, dt) or target.distance
                    update_target_info(target.size, distNow)
                    DistLabel.Text = "Dist: " .. math.floor(distNow)

                    if not is_ghost_alive(target) then
                        task.wait(0.3)
                    end
                else
                    update_target_info(nil)
                    DistLabel.Text = "Dist: --"
                    task.wait(0.5)
                end
            end)

            if not okIter then
                running = false
                set_farm_ui(false, "Error! См. консоль", Color3.fromRGB(255, 140, 60))
                warn("[GhostFarm] " .. tostring(errIter))
                return
            end
        end
    end

    -- ============================================
    -- ДЕЙСТВИЯ
    -- ============================================

    local function toggle_farm()
        running = not running
        activeLoopId = activeLoopId + 1

        if running then
            set_farm_ui(true, "Running...", GREEN)
            task.spawn(farm_loop, activeLoopId)
        else
            set_farm_ui(false, "Stopped", RED)
        end
    end

    local function prev_target()
        autoTarget = false
        AutoButton.Text = "Auto Target: OFF"
        AutoButton.TextColor3 = Color3.fromRGB(255, 100, 100)

        local ghosts = get_alive_ghosts()
        if #ghosts > 0 then
            currentTargetIndex = currentTargetIndex - 1
            if currentTargetIndex < 1 then
                currentTargetIndex = #ghosts
            end
        end
    end

    local function next_target()
        autoTarget = false
        AutoButton.Text = "Auto Target: OFF"
        AutoButton.TextColor3 = Color3.fromRGB(255, 100, 100)

        local ghosts = get_alive_ghosts()
        if #ghosts > 0 then
            currentTargetIndex = currentTargetIndex + 1
            if currentTargetIndex > #ghosts then
                currentTargetIndex = 1
            end
        end
    end

    -- ============================================
    -- КНОПКИ И СЛАЙДЕР
    -- ============================================

    ToggleButton.MouseButton1Click:Connect(function()
        toggle_farm()
    end)

    AutoButton.MouseButton1Click:Connect(function()
        autoTarget = not autoTarget
        if autoTarget then
            currentTargetIndex = 1
            AutoButton.Text = "Auto Target: ON"
            AutoButton.TextColor3 = Color3.fromRGB(100, 255, 100)
        else
            AutoButton.Text = "Auto Target: OFF"
            AutoButton.TextColor3 = Color3.fromRGB(255, 100, 100)
        end
    end)

    PrevButton.MouseButton1Click:Connect(prev_target)
    NextButton.MouseButton1Click:Connect(next_target)

    -- Горячие клавиши
    track(UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end

        if input.KeyCode == Enum.KeyCode.F4 then
            toggle_farm()
        elseif input.KeyCode == Enum.KeyCode.F5 then
            prev_target()
        elseif input.KeyCode == Enum.KeyCode.F6 then
            next_target()
        end
    end))

    -- Слайдер скорости
    local draggingSpeed = false

    local function isPointer(t)
        return t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch
    end

    local function set_speed_from_x(x)
        local absPos = SpeedBar.AbsolutePosition
        local absSize = SpeedBar.AbsoluteSize
        if absSize.X <= 0 then return end

        local rel = math.clamp((x - absPos.X) / absSize.X, 0, 1)
        flySpeed = math.floor(20 + rel * 130)
        SpeedFill.Size = UDim2.new(rel, 0, 1, 0)
        SpeedKnob.Position = UDim2.new(rel, -7, 0.5, -7)
        SpeedLabel.Text = "Fly Speed: " .. flySpeed
    end

    SpeedBar.InputBegan:Connect(function(input)
        if isPointer(input.UserInputType) then
            draggingSpeed = true
            set_speed_from_x(input.Position.X)
        end
    end)

    track(UserInputService.InputChanged:Connect(function(input)
        if draggingSpeed and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            set_speed_from_x(input.Position.X)
        end
    end))

    track(UserInputService.InputEnded:Connect(function(input)
        if isPointer(input.UserInputType) then
            draggingSpeed = false
        end
    end))

    print("[Farm] v1.4 GUI загружен! F4 - Start/Stop")
end)

if not okMain then
    warn("[GhostFarm] Ошибка загрузки: " .. tostring(errMain))
end
