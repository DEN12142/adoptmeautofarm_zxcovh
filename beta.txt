    local function get_alive_ghosts()
        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return {} end

        local visuals = workspace:FindFirstChild("GhostClustersVisuals")
        if not visuals then return {} end

        local ghosts = {}
        local playerPos = hrp.Position

        for _, model in ipairs(visuals:GetChildren()) do
            if model:IsA("Model") then
                local ghostId = model:GetAttribute(GHOST_ATTR.Id)
                if ghostId then
                    local isShielded = model:GetAttribute(GHOST_ATTR.Shielded) == true
                    local isOrbiting = model:GetAttribute(GHOST_ATTR.Orbiting) == true
                    local progress = model:GetAttribute(GHOST_ATTR.Progress) or 0
                    local required = model:GetAttribute(GHOST_ATTR.ProgressRequired) or 1

                    if not isShielded and not isOrbiting and progress < required then
                        local pos = model:GetPivot().Position
                        local dx = playerPos.X - pos.X
                        local dz = playerPos.Z - pos.Z
                        local dist = math.sqrt(dx * dx + dz * dz)

                        table.insert(ghosts, {
                            model = model,
                            id = ghostId,
                            size = model:GetAttribute(GHOST_ATTR.Size),
                            position = pos,
                            distance = dist,
                        })
                    end
                end
            end
        end

        table.sort(ghosts, function(a, b)
            return a.distance < b.distance
        end)

        return ghosts
    end

    local function fly_to(targetPos, dt)
        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return nil end

        local currentPos = hrp.Position
        local direction = targetPos - currentPos
        local distance = direction.Magnitude

        if distance >= 3 then
            local moveVector = direction.Unit * math.min(flySpeed * dt, distance)
            hrp.CFrame = hrp.CFrame + moveVector
            hrp.Velocity = Vector3.new(0, 0, 0)
            hrp.RotVelocity = Vector3.new(0, 0, 0)
        end

        local p = hrp.Position
        local dx = targetPos.X - p.X
        local dz = targetPos.Z - p.Z
        return math.sqrt(dx * dx + dz * dz)
    end

    -- Наводим персонажа и камеру на призрака
    local function aim_at(targetPos)
        local camera = workspace.CurrentCamera
        if not camera then return end

        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local camPos = camera.CFrame.Position
        local lookDist = (targetPos - camPos).Magnitude
        if lookDist < 0.05 then return end

        -- Камера смотрит на призрака (от позиции камеры, без смещения)
        camera.CFrame = CFrame.lookAt(camPos, targetPos)

        -- Персонаж поворачивается в сторону призрака (только yaw)
        local flatDir = Vector3.new(targetPos.X - hrp.Position.X, 0, targetPos.Z - hrp.Position.Z)
        if flatDir.Magnitude > 0.5 then
            local yaw = math.atan(-flatDir.X, -flatDir.Z)
            hrp.CFrame = CFrame.new(hrp.Position) * CFrame.Angles(0, yaw, 0)
            hrp.Velocity = Vector3.new(0, hrp.Velocity.Y, 0)
            hrp.RotVelocity = Vector3.new(0, 0, 0)
        end
    end

    -- Ищем функции нажатия мыши (в _G и напрямую)
    local function find_fn(...)
        local names = { ... }
        for _, name in ipairs(names) do
            local ok, fn = pcall(function()
                local v = _G[name]
                if type(v) == "function" then return v end
                return nil
            end)
            if ok and fn then return fn, name end
        end
        -- пробуем прямое имя (если экзекутор положил в env, а не в _G)
        for _, name in ipairs(names) do
            local ok, fn = pcall(function()
                return loadstring("return " .. name)()
            end)
            if ok and type(fn) == "function" then return fn, name end
        end
        return nil
    end

    local mousePressFn, pressName = find_fn("mouse1press", "m1press", "mouse1_down", "Button1Down")

        local ok, wrapper = pcall(function()

            local tools = UIManager.apps.ToolApp.active_tools

            for _, w in ipairs(snapshot) do

            end
            return nil

        if ok then return wrapper end
        return nil
    end

    local function set_lmb(held)
        if held == lmbHeld then return end

        if held then
            local ok = false

            -- Метод 1: нажатие мыши экзекутором
            if mousePressFn then
                ok = pcall(mousePressFn)
                if ok then lmbMethod = "mouse" end
            end

            -- Метод 2: вызов use() у обёртки инструмента игры
            -- (блокируется до stop — запускаем в корутине)
            if not ok then
                local wrapper = get_vacuum_tool()
                if wrapper and type(wrapper.use) == "function" then
                    local used = false
                    sessionThread = task.spawn(function()
                        pcall(function()
                            wrapper:use({
                                source = "AutoFarm",
                                aim_mode = "auto",
                            })
                        end)
                    end)
                    used = sessionThread ~= nil
                    if used then lmbMethod = "session" end
                    ok = used
                end
            end

            if not ok then
                lmbMethod = nil
                lmbHeld = false
                lmbFailed = true
                update_lmb_label()
                warn("[GhostFarm] LMB не сработал: mousePressFn=" .. tostring(mousePressFn) .. ", tool=" .. tostring(get_vacuum_tool()))
                return
            end

            lmbHeld = true
            lmbFailed = false
            update_lmb_label()
        else
            if lmbMethod == "mouse" and mouseReleaseFn then
                pcall(mouseReleaseFn)
            elseif lmbMethod == "session" then
                local w = get_vacuum_tool()
                local s = w and w.ghost_clusters_session
                if s then
                    pcall(function() s:stop() end)
                end
                if w and type(w.cancel_active_use) == "function" then
                    pcall(function() w:cancel_active_use() end)
                end
            end
            lmbHeld = false
            lmbMethod = nil
            lmbFailed = false
            update_lmb_label()
        end
    end

    local function release_lmb()
        if not lmbHeld then return end
        set_lmb(false)
    end

    -- Для режима "cursor" игра целится по позиции мыши — двигаем курсор на призрака
    local function aim_cursor(targetPos)
        if not mouseMoveFn or lmbMethod ~= "mouse" then return end
        local camera = workspace.CurrentCamera
        if not camera then return end
        local screen, onScreen = camera:WorldToViewportPoint(targetPos)
        if not onScreen then return end
        local mouse = UserInputService:GetMouseLocation()
        local dx = screen.X - mouse.X
        local dy = screen.Y - mouse.Y
        if math.abs(dx) > 2 or math.abs(dy) > 2 then
            pcall(mouseMoveFn, dx, dy)
        end
    end

    -- Игра сама может остановить пылесос (атака призрака -> _stop_vacuum),
    -- а наш флаг остаётся "зажатым" -> периодически перенажимаем
    local function ensure_lmb()
        if not lmbHeld then return end

        local now = os.clock()

        if lmbMethod == "session" then
            if now - lastMouseRefresh < 0.5 then return end
            lastMouseRefresh = now
            if not is_sucking() then
                lmbHeld = false
                sessionThread = nil
                set_lmb(true)
            end
        elseif lmbMethod == "mouse" then
            if now - lastMouseRefresh < 1.0 then return end
            lastMouseRefresh = now
            -- короткое отпускание+нажатие, чтобы игра увидела новое InputBegan
            if mouseReleaseFn then pcall(mouseReleaseFn) end
            if mousePressFn then
                pcall(mousePressFn)
            end
        end
    end

    local function setup_anti_stun()
        if antiStunHooked then return true end
        local ok, mod = pcall(function()
            local p = ReplicatedStorage:WaitForChild("ClientModules", 5)
            p = p and p:WaitForChild("Game", 5)
            p = p and p:WaitForChild("MinigameHelpers", 5)

            if not p then return nil end

        end)
        if not ok or type(mod) ~= "table" then
            warn("[GhostFarm] Anti-Stun: MinigameRagdoll не загрузился: " .. tostring(mod))
            return false
        end

        local hooked = pcall(function()
            if type(mod.fling) == "function" then
                local oldFling = mod.fling
                mod.fling = function(...)
                    if antiStun then return end
                    return oldFling(...)
                end
            end
            if type(mod.enable) == "function" then
                local oldEnable = mod.enable
                mod.enable = function(tag, ...)
                    if antiStun and tag == "ghost_clusters_tornado" then return end
                    return oldEnable(tag, ...)
                end
            end
        end)
        if not hooked then
            warn("[GhostFarm] Anti-Stun: не удалось переопределить функции MinigameRagdoll")
            return false
        end

        antiStunHooked = true
        return true
    end

    setup_anti_stun()

    -- Сбрасываем врезавшуюся в персонажа торнадо-силу
    local function clear_tornado_force()
        local character = Player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local force = hrp:FindFirstChild("GhostClustersTornadoForce")
        if force then
            pcall(function() force:Destroy() end)
        end
        local att = hrp:FindFirstChild("GhostClustersTornadoAttachment")
        if att then
            pcall(function() att:Destroy() end)
        end
        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end

    -- ФАРМ ЦИКЛ
    -- ============================================

    local function farm_loop(myLoopId)
        local errCount = 0

        while running and myLoopId == activeLoopId and _G.GCF_TOKEN == myToken do
            local waited = task.wait(0.03)
            local dt = (type(waited) == "number") and math.clamp(waited, 0.01, 0.1) or 0.03

            if not (running and myLoopId == activeLoopId and _G.GCF_TOKEN == myToken) then
                break
            end

            local okIter, errIter = pcall(function()
                if not antiStunHooked then
                    setup_anti_stun()
                end
                if antiStun then
                    clear_tornado_force()
                end

                local ghosts = get_alive_ghosts()
                GhostsLabel.Text = "Ghosts: " .. #ghosts

                local target
                if autoTarget then
                    target = ghosts[1]
                elseif #ghosts > 0 then
                    if currentTargetIndex > #ghosts then
                        currentTargetIndex = 1
                    end
                    target = ghosts[currentTargetIndex]
                end

                if target then
                    local distNow = fly_to(target.position, dt) or target.distance
                    aim_at(target.position)
                    aim_cursor(target.position)
                    set_lmb(true)
                    ensure_lmb()
                    update_target_info(target.size, distNow)
                    DistLabel.Text = "Dist: " .. math.floor(distNow)

                    if StatusLabel.Text ~= "Running..." then
                        update_status("Running...", GREEN)
                    end

                    if not is_ghost_alive(target) then
                        task.wait(0.3)
                    end
                else
                    set_lmb(false)
                    update_target_info(nil)
                    DistLabel.Text = "Dist: --"

                    -- Ждём начала игры (призраки ещё не заспавнились)
                    if StatusLabel.Text ~= "Waiting for game..." then
                        update_status("Waiting for game...", Color3.fromRGB(255, 200, 60))
                    end

                    task.wait(0.5)
                end
            end)

            if not okIter then
                -- Не убиваем цикл: ждём и пробуем снова (ошибка могла быть временной)
                errCount = errCount + 1
                warn("[GhostFarm] ошибка #" .. errCount .. ": " .. tostring(errIter))
                update_status("Error x" .. errCount .. " (retry...)", Color3.fromRGB(255, 140, 60))
                task.wait(1)
            else
                errCount = 0
            end
        end

        release_lmb()
    end

    local function restore_camera()
        pcall(function()
            workspace.CurrentCamera.CameraType = Enum.CameraType.Custom

    local function toggle_farm()
        running = not running
        activeLoopId = activeLoopId + 1

        if running then
            pcall(function()
                workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable
            end)
            set_farm_ui(true, "Running...", GREEN)
            task.spawn(farm_loop, activeLoopId)
        else
            set_farm_ui(false, "Stopped", RED)
            release_lmb()
            restore_camera()
        end
    end

    local function prev_target()
        autoTarget = false
        AutoButton.Text = "Auto Target: OFF"
        AutoButton.TextColor3 = Color3.fromRGB(255, 100, 100)

        local ghosts = get_alive_ghosts()
        if #ghosts > 0 then
            currentTargetIndex = currentTargetIndex - 1
            if currentTargetIndex < 1 then
                currentTargetIndex = #ghosts
            end
        end
    end

    local function next_target()
        autoTarget = false
        AutoButton.Text = "Auto Target: OFF"
        AutoButton.TextColor3 = Color3.fromRGB(255, 100, 100)

        local ghosts = get_alive_ghosts()
        if #ghosts > 0 then
            currentTargetIndex = currentTargetIndex + 1
            if currentTargetIndex > #ghosts then
                currentTargetIndex = 1
            end
        end
    end

    -- Слайдер скорости
    local draggingSpeed = false

    local function isPointer(t)
        return t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch
    end

    local function set_speed_from_x(x)
        local absPos = SpeedBar.AbsolutePosition
        local absSize = SpeedBar.AbsoluteSize
        if absSize.X <= 0 then return end

        local rel = math.clamp((x - absPos.X) / absSize.X, 0, 1)
        flySpeed = math.floor(20 + rel * 130)
        SpeedFill.Size = UDim2.new(rel, 0, 1, 0)
        SpeedKnob.Position = UDim2.new(rel, -7, 0.5, -7)
        SpeedLabel.Text = "Fly Speed: " .. flySpeed
    end

    SpeedBar.InputBegan:Connect(function(input)
        if isPointer(input.UserInputType) then
            draggingSpeed = true
            set_speed_from_x(input.Position.X)
        end
    end)

    track(UserInputService.InputChanged:Connect(function(input)
        if draggingSpeed and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            set_speed_from_x(input.Position.X)
        end
    end))

    track(UserInputService.InputEnded:Connect(function(input)
        if isPointer(input.UserInputType) then
            draggingSpeed = false
        end
    end))
